% ── Typography ──────────────────────────────────────────────
\usepackage{fontspec}
\setmainfont{TeX Gyre Pagella}[
  Numbers=OldStyle,
  Ligatures=TeX
]
\setmonofont{DejaVu Sans Mono}[Scale=0.82]
\usepackage{unicode-math}
\setmathfont{TeX Gyre Pagella Math}
\usepackage{microtype}
\linespread{1.08}

% ── Page geometry ──────────────────────────────────────────
\usepackage[
  letterpaper,
  top=1.15in, bottom=1.15in,
  left=1.35in, right=1.35in
]{geometry}

% ── Headers / footers ─────────────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\small\thepage}

% ── Theorem environments ───────────────────────────────────
\usepackage{amsthm}
\newtheoremstyle{gse}%           name
  {10pt}%                        space above
  {8pt}%                         space below
  {\itshape}%                    body font
  {}%                            indent
  {\bfseries}%                   head font
  {.}%                           punctuation after head
  {.5em}%                        space after head
  {}%                            head spec
\theoremstyle{gse}
\newtheorem{theorem}{Theorem}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}
\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}
\theoremstyle{remark}
\newtheorem*{remark}{Remark}

% ── Section title formatting ──────────────────────────────
\usepackage{titlesec}
\titleformat{\section}
  {\normalfont\large\bfseries}{\thesection.}{0.5em}{}
\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}{\thesubsection}{0.5em}{}
\titlespacing*{\section}{0pt}{18pt plus 4pt minus 2pt}{8pt plus 2pt minus 1pt}
\titlespacing*{\subsection}{0pt}{12pt plus 3pt minus 1pt}{4pt plus 1pt minus 1pt}

% ── Abstract formatting ───────────────────────────────────
\renewenvironment{abstract}{%
  \small
  \begin{center}
    {\bfseries\abstractname\vspace{-0.5em}\vspace{0pt}}
  \end{center}
  \quotation
}{\endquotation}

% ── Math extras ────────────────────────────────────────────
\usepackage{amsmath,amssymb}
\usepackage{mathtools}
\DeclareMathOperator{\Vol}{Vol}
\DeclareMathOperator{\Ric}{Ric}

% ── Table formatting ──────────────────────────────────────
\usepackage{booktabs}
\usepackage{array}

% ── Code / monospace ──────────────────────────────────────
\usepackage{listings}
\lstset{
  basicstyle=\small\ttfamily,
  breaklines=true,
  frame=none
}

% ── Hyperlinks (pandoc loads hyperref; just set options) ──

  \hypersetup{
    colorlinks=true,
    linkcolor=black,
    citecolor=black,
    urlcolor=black!60!blue
  }%
}
